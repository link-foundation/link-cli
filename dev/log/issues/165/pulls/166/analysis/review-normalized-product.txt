#!/usr/bin/env node

/**
 * Check if a C# release is needed based on changesets and NuGet registry state.
 *
 * This script checks:
 *   1. If there are changeset files to process.
 *   2. If the current csproj <Version> has already been published to NuGet.
 *
 * IMPORTANT: This script checks NuGet (the source of truth for the package),
 * NOT git tags. Git tags can exist without the package being on NuGet — that is
 * exactly the failure mode that caused issue #84: a tag was pushed but
 * `dotnet nuget push` returned HTTP 403, so the public package was never
 * created. Re-runs then short-circuited on the tag and skipped publish.
 *
 * This provides a self-healing mechanism: if a previous release attempt failed
 * after the version commit and tag were pushed, the next push to main detects
 * the unpublished version on NuGet and resumes the publish + GitHub release
 * steps without requiring a new changeset.
 *
 * Analogous to `check-release-needed.mjs` in the JavaScript template and
 * `check-release-needed.rs` in the Rust template.
 *
 * Usage: node csharp/scripts/check-release-needed.mjs
 *   [--csproj <path>] [--repository <owner/repo>] [--tag-prefix <prefix>]
 *
 * Environment variables:
 *   - HAS_CHANGESETS:    'true' if changeset files exist (from check_changesets step)
 *   - NUGET_INDEX_URL:   override NuGet flat-container endpoint (for tests)
 *   - GITHUB_API_URL:    override GitHub API endpoint (for tests)
 *   - GITHUB_REPOSITORY: owner/repo, used when --repository is omitted
 *
 * Outputs (written to GITHUB_OUTPUT):
 *   - should_release:        'true' if a release should be created
 *   - skip_bump:             'true' if the version bump should be skipped
 *                            (csproj version is not yet on NuGet)
 *   - current_version:       the current csproj <Version> value
 *   - nuget_published:       'true' if the current version is on NuGet
 *   - github_release_exists: 'true' if the matching GitHub release exists
 *   - reason:                short human-readable reason
 *
 * Addresses the same defect class documented in
 * link-foundation/js-ai-driven-development-pipeline-template issue #36.
 */

import { appendFileSync, readFileSync } from 'node:fs';
import { setTimeout as sleep } from 'node:timers/promises';

const NUGET_FLAT_CONTAINER = 'https://api.nuget.org/v3-flatcontainer';
const GITHUB_API = 'https://api.github.com';

const args = process.argv.slice(2);

/**
 * Read a CLI flag value.
 * @param {string} name
 * @returns {string | null}
 */
function getArg(name) {
  const equalsIndex = args.findIndex((arg) => arg.startsWith(`--${name}=`));
  if (equalsIndex !== -1) {
    return args[equalsIndex].slice(`--${name}=`.length);
  }
  const index = args.indexOf(`--${name}`);
  if (index === -1) {
    return null;
  }
  const value = args[index + 1];
  if (value === undefined || value.startsWith('--')) {
    return '';
  }
  return value;
}

/**
 * Append a key/value pair to GITHUB_OUTPUT (when defined) and echo to stdout.
 * @param {string} key
 * @param {string} value
 */
export function setOutput(key, value) {
  const outputFile = process.env.GITHUB_OUTPUT;
  if (outputFile) {
    appendFileSync(outputFile, `${key}=${value}\n`);
  }
  console.log(`Output: ${key}=${value}`);
}

/**
 * Append a markdown block to GITHUB_STEP_SUMMARY (when defined).
 * @param {string} markdown
 */
function appendStepSummary(markdown) {
  const summaryFile = process.env.GITHUB_STEP_SUMMARY;
  if (summaryFile) {
    appendFileSync(summaryFile, `${markdown}\n`);
  }
}

/**
 * Extract <Version> and <PackageId> from a csproj file.
 * @param {string} csprojPath
 * @returns {{ version: string, packageId: string }}
 */
export function readCsprojInfo(csprojPath) {
  const content = readFileSync(csprojPath, 'utf-8');

  const versionMatch = content.match(/<Version>([^<]+)<\/Version>/);
  if (!versionMatch) {
    throw new Error(`Could not parse <Version> from ${csprojPath}`);
  }

  const packageIdMatch = content.match(/<PackageId>([^<]+)<\/PackageId>/);
  // PackageId falls back to the AssemblyName/csproj file name when omitted —
  // expose the explicit value to the caller, otherwise the workflow will
  // resolve via msbuild like it already does for the publish step.
  return {
    version: versionMatch[1].trim(),
    packageId: packageIdMatch ? packageIdMatch[1].trim() : '',
  };
}

/**
 * `fetchImpl(url, options)`, retried with backoff on network failures, 429
 * and 5xx, so one dropped connection does not fail the release decision on
 * main (issue #134). Other answers are returned at once; after the last
 * attempt the last answer is returned or the network error is thrown with
 * its cause, which Node's "fetch failed" message leaves out.
 */
export async function fetchWithRetries(
  url,
  options,
  fetchImpl = fetch,
  { attempts = 4, delayMs = 1000 } = {},
) {
  for (let attempt = 1; ; attempt += 1) {
    let response;
    try {
      response = await fetchImpl(url, options);
    } catch (error) {
      if (attempt >= attempts) {
        const cause = error.cause
          ? `: ${error.cause.code ?? error.cause.message ?? error.cause}`
          : '';
        throw new Error(`${url}: ${error.message}${cause} (after ${attempt} attempts)`);
      }
    }
    const transient = !response || response.status === 429 || response.status >= 500;
    if (!transient || attempt >= attempts) return response;
    console.log(`Retrying ${url} (attempt ${attempt + 1} of ${attempts})`);
    await sleep(delayMs * 2 ** (attempt - 1));
  }
}

/**
 * Probe `https://api.nuget.org/v3-flatcontainer/{id-lower}/index.json` for the
 * package's published versions.
 *
 * Returns null when the package id is not registered on NuGet at all (HTTP 404
 * for the index endpoint). Returns an empty array if the registration exists
 * but the index has no versions (extremely rare). Otherwise returns the
 * declared version list.
 *
 * @param {string} packageId
 * @returns {Promise<string[] | null>}
 */
export async function fetchNugetVersions(packageId, fetchImpl = fetch, retry = {}) {
  const baseUrl = process.env.NUGET_INDEX_URL ?? NUGET_FLAT_CONTAINER;
  const url = `${baseUrl}/${packageId.toLowerCase()}/index.json`;
  console.log(`Fetching ${url}`);

  const response = await fetchWithRetries(url, undefined, fetchImpl, retry);
  if (response.status === 404) {
    return null;
  }
  if (!response.ok) {
    throw new Error(`NuGet flat-container index returned HTTP ${response.status} for ${packageId}`);
  }
  const payload = await response.json();
  return Array.isArray(payload.versions) ? payload.versions : [];
}

/**
 * Probe `GET /repos/{owner}/{repo}/releases/tags/{tag}` to see if a matching
 * GitHub release already exists.
 *
 * @param {string} repository  owner/repo
 * @param {string} tag         full tag, e.g. csharp-v2.4.0
 * @returns {Promise<boolean>}
 */
export async function fetchGithubReleaseExists(repository, tag, fetchImpl = fetch, retry = {}) {
  if (!repository) {
    return false;
  }
  const baseUrl = process.env.GITHUB_API_URL ?? GITHUB_API;
  const url = `${baseUrl}/repos/${repository}/releases/tags/${encodeURIComponent(tag)}`;
  console.log(`Fetching ${url}`);

  const headers = { Accept: 'application/vnd.github+json' };
  if (process.env.GH_TOKEN) {
    headers.Authorization = `Bearer ${process.env.GH_TOKEN}`;
  } else if (process.env.GITHUB_TOKEN) {
    headers.Authorization = `Bearer ${process.env.GITHUB_TOKEN}`;
  }

  const response = await fetchWithRetries(url, { headers }, fetchImpl, retry);
  if (response.status === 404) {
    return false;
  }
  if (!response.ok) {
    throw new Error(`GitHub releases endpoint returned HTTP ${response.status}`);
  }
  return true;
}

/**
 * @typedef {object} CheckResult
 * @property {boolean} shouldRelease
 * @property {boolean} skipBump
 * @property {string}  currentVersion
 * @property {boolean} nugetPublished
 * @property {boolean} githubReleaseExists
 * @property {string}  reason
 */

/**
 * Pure decision function — exported for unit tests.
 * @param {object} input
 * @param {boolean} input.hasChangesets
 * @param {string}  input.currentVersion
 * @param {string[] | null} input.publishedVersions
 *   `null` when the package id is not registered on NuGet at all.
 * @param {boolean} input.githubReleaseExists
 * @returns {CheckResult}
 */
export function decide({ hasChangesets, currentVersion, publishedVersions, githubReleaseExists }) {
  const nugetPublished =
    Array.isArray(publishedVersions) && publishedVersions.includes(currentVersion);

  if (hasChangesets) {
    return {
      shouldRelease: true,
      skipBump: false,
      currentVersion,
      nugetPublished,
      githubReleaseExists,
      reason: 'changesets present — normal release path',
    };
  }

  if (!nugetPublished) {
    return {
      shouldRelease: true,
      skipBump: true,
      currentVersion,
      nugetPublished,
      githubReleaseExists,
      reason:
        publishedVersions === null
          ? `package not yet registered on NuGet — self-healing resume for v${currentVersion}`
          : `v${currentVersion} not yet published on NuGet — self-healing resume`,
    };
  }

  if (!githubReleaseExists) {
    return {
      shouldRelease: true,
      skipBump: true,
      currentVersion,
      nugetPublished,
      githubReleaseExists,
      reason: `v${currentVersion} on NuGet but no GitHub release — self-healing release creation`,
    };
  }

  return {
    shouldRelease: false,
    skipBump: false,
    currentVersion,
    nugetPublished,
    githubReleaseExists,
    reason: `v${currentVersion} already on NuGet and GitHub — no release needed`,
  };
}

async function main() {
  const csprojPath =
    getArg('csproj') || 'csharp/Foundation.Data.Doublets.Cli/Foundation.Data.Doublets.Cli.csproj';
  const repository = getArg('repository') || process.env.GITHUB_REPOSITORY || '';
  const tagPrefix = getArg('tag-prefix') || 'csharp-v';
  const packageIdOverride = getArg('package-id') || '';

  const csproj = readCsprojInfo(csprojPath);
  const packageId = packageIdOverride || csproj.packageId || 'clink';
  const currentVersion = csproj.version;

  console.log(`csproj path:     ${csprojPath}`);
  console.log(`Package id:      ${packageId}`);
  console.log(`Current version: ${currentVersion}`);
  console.log(`Repository:      ${repository || '(not set)'}`);
  console.log(`Has changesets:  ${process.env.HAS_CHANGESETS === 'true'}`);

  const publishedVersions = await fetchNugetVersions(packageId);
  if (publishedVersions === null) {
    console.log(`NuGet: package "${packageId}" not registered yet`);
  } else {
    console.log(`NuGet: ${publishedVersions.length} version(s) registered`);
    console.log(`NuGet versions: ${publishedVersions.join(', ')}`);
  }

  const tag = `${tagPrefix}${currentVersion}`;
  const githubReleaseExists = await fetchGithubReleaseExists(repository, tag);
  console.log(`GitHub release ${tag}: ${githubReleaseExists ? 'exists' : 'missing'}`);

  const decision = decide({
    hasChangesets: process.env.HAS_CHANGESETS === 'true',
    currentVersion,
    publishedVersions,
    githubReleaseExists,
  });

  console.log(`Decision: ${decision.reason}`);

  setOutput('should_release', decision.shouldRelease ? 'true' : 'false');
  setOutput('skip_bump', decision.skipBump ? 'true' : 'false');
  setOutput('current_version', decision.currentVersion);
  setOutput('nuget_published', decision.nugetPublished ? 'true' : 'false');
  setOutput('github_release_exists', decision.githubReleaseExists ? 'true' : 'false');
  setOutput('reason', decision.reason);

  appendStepSummary(
    `### C# release decision\n\n` +
      `- Package: \`${packageId}\`\n` +
      `- csproj \`<Version>\`: \`${currentVersion}\`\n` +
      `- On NuGet: ${decision.nugetPublished ? 'yes' : 'no'}\n` +
      `- GitHub release \`${tag}\`: ${decision.githubReleaseExists ? 'exists' : 'missing'}\n` +
      `- \`should_release\`: \`${decision.shouldRelease}\`\n` +
      `- \`skip_bump\`: \`${decision.skipBump}\`\n` +
      `- Reason: ${decision.reason}\n`,
  );
}

// Allow `import { decide, readCsprojInfo, ... }` without running main().
const entryPath = process.argv[1];
const invokedDirectly =
  typeof entryPath === 'string' &&
  entryPath.length > 0 &&
  (import.meta.url === `file://${entryPath}` || import.meta.url.endsWith(entryPath));
if (invokedDirectly) {
  main().catch((error) => {
    console.error('Error:', error.message);
    process.exit(1);
  });
}

/**
 * Build a work clone with a bare origin holding a minimal C# project and one
 * changeset, as the release job sees it after checkout.
 */
function createCsharpReleaseSandbox(prefix) {
  const dir = mkdtempSync(join(tmpdir(), prefix));
  const remote = join(dir, 'remote.git');
  const work = join(dir, 'work');
  mkdirSync(work, { recursive: true });
  runGit(['init', '--bare', '--initial-branch=main', remote], dir);
  runGit(['init', '-b', 'main'], work);
  runGit(['config', 'user.name', 'Test User'], work);
  runGit(['config', 'user.email', 'test@example.com'], work);
  const projectDir = join(work, 'csharp', 'Foundation.Data.Doublets.Cli');
  const changesetDir = join(work, 'csharp', '.changeset');
  mkdirSync(projectDir, { recursive: true });
  mkdirSync(changesetDir, { recursive: true });
  writeFileSync(
    join(projectDir, 'Foundation.Data.Doublets.Cli.csproj'),
    '<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><Version>2.3.0</Version></PropertyGroup></Project>\n',
  );
  writeFileSync(
    join(changesetDir, 'release.md'),
    "---\n'Foundation.Data.Doublets.Cli': patch\n---\n\nFix release automation.\n",
  );
  runGit(['add', '.'], work);
  runGit(['commit', '-m', 'initial'], work);
  runGit(['remote', 'add', 'origin', remote], work);
  runGit(['push', '-u', 'origin', 'main'], work);
  return { dir, remote, work, outputFile: join(dir, 'github-output.txt') };
}

  writeFileSync(path, `#!/bin/sh\n${body}\n`);
  chmodSync(path, 0o755);
}

test('version-and-commit rebases and retries when another release pushed first', () => {
  const { dir, remote, work, outputFile } = createCsharpReleaseSandbox('link-cli-csharp-race-');
  const rival = join(dir, 'rival');
  runGit(['clone', remote, rival], dir);
  runGit(['config', 'user.name', 'Rival'], rival);
  runGit(['config', 'user.email', 'rival@example.com'], rival);
  // Right after the release commit is created, a concurrent workflow (e.g. the
  // Rust release) lands its own commit on main, so the first push loses the race.
  const marker = join(dir, 'raced');
  writeHook(
    join(work, '.git', 'hooks', 'post-commit'),
    `[ -e "${marker}" ] && exit 0\ntouch "${marker}"\n` +
      `cd "${rival}" && echo rust > rust.txt && git add rust.txt && ` +
      `git commit -q -m "chore: release rust-v1.0.9" && git push -q origin main`,
  );

  const stdout = runNode('csharp/scripts/version-and-commit.mjs', ['--mode', 'changeset'], {
    cwd: work,
    env: { GITHUB_OUTPUT: outputFile },
  });

  assert.ok(existsSync(marker), 'the concurrent push must have happened');
  assert.match(stdout, /Push rejected as non-fast-forward \(attempt 1\/3\)/);
  assert.match(stdout, /Pushed release commit and tag csharp-v2\.3\.1/);
  const log = runGit(['--git-dir', remote, 'log', '--format=%s', 'main'], dir).trim().split('\n');
  assert.deepEqual(log, ['chore(csharp): release v2.3.1', 'chore: release rust-v1.0.9', 'initial']);
  // The tag points at the rebased commit that actually reached main.
  assert.equal(
    runGit(['--git-dir', remote, 'rev-parse', 'csharp-v2.3.1^{commit}'], dir),
    runGit(['--git-dir', remote, 'rev-parse', 'main'], dir),
  );
  assert.equal(runGit(['status', '--porcelain'], work), '');
  assert.match(readFileSync(outputFile, 'utf8'), /^version_committed=true$/m);
});

function startNugetAndGithubMock({ versions, githubReleaseStatus }) {
  const sockets = new Set();
  const server = createServer((req, res) => {
    res.setHeader('connection', 'close');
    if (req.url?.startsWith('/nuget/')) {
      if (versions === null) {
        res.writeHead(404).end();
      } else {
        res.writeHead(200, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ versions }));
      }
      return;
    }
    if (req.url?.startsWith('/github/')) {
      res.writeHead(githubReleaseStatus).end();
      return;
    }
    res.writeHead(500).end();
  });
  server.on('connection', (socket) => {
    sockets.add(socket);
    socket.on('close', () => sockets.delete(socket));
  });

  return new Promise((resolve) => {
    server.listen(0, '127.0.0.1', () => {
      const { port } = server.address();
      resolve({
        nugetUrl: `http://127.0.0.1:${port}/nuget`,
        githubUrl: `http://127.0.0.1:${port}/github`,
        close: () =>
          new Promise((r) => {
            for (const socket of sockets) {
              socket.destroy();
            }
            server.close(() => r());
          }),
      });
    });
  });
}

// Plays back `answers`: an Error is thrown like a network failure, a number
// is answered as that HTTP status.
function playback(answers, body = {}) {
  const calls = [];
  const fetchImpl = async (url) => {
    calls.push(url);
    const answer = answers[calls.length - 1];
    if (answer instanceof Error) throw answer;
    return { ok: answer >= 200 && answer < 300, status: answer, json: async () => body };
  };
  return { calls, fetchImpl };
}

// execFileSync forwards a child's stderr to ours unless `stdio` is set. On a
// runner that turned the `::error` line of the expected-failure test into a
// real error annotation on a green run (issue #134), so the children's output
// is always captured; a failing call still exposes it as `error.stderr`.
function runNode(script, args, options = {}) {
  return execFileSync(process.execPath, [join(repoRoot, script), ...args], {
    cwd: options.cwd ?? repoRoot,
    encoding: 'utf8',
    stdio: 'pipe',
    env: { ...process.env, ...(options.env ?? {}) },
  });
}

async function runNodeAsync(script, args, options = {}) {
  const { stdout } = await execFileAsync(process.execPath, [join(repoRoot, script), ...args], {
    cwd: options.cwd ?? repoRoot,
    encoding: 'utf8',
    env: { ...process.env, ...(options.env ?? {}) },
  });
  return stdout;
}

function runGit(args, cwd) {

function startNugetAndGithubMock({ versions, githubReleaseStatus }) {
  const sockets = new Set();
  const server = createServer((req, res) => {
    res.setHeader('connection', 'close');
    if (req.url?.startsWith('/nuget/')) {
      if (versions === null) {
        res.writeHead(404).end();
      } else {
        res.writeHead(200, { 'content-type': 'application/json' });
        res.end(JSON.stringify({ versions }));
      }
      return;
    }
    if (req.url?.startsWith('/github/')) {
      res.writeHead(githubReleaseStatus).end();
      return;
    }
    res.writeHead(500).end();
  });
  server.on('connection', (socket) => {
    sockets.add(socket);
    socket.on('close', () => sockets.delete(socket));
  });

  return new Promise((resolve) => {
    server.listen(0, '127.0.0.1', () => {
      const { port } = server.address();
      resolve({
        nugetUrl: `http://127.0.0.1:${port}/nuget`,
        githubUrl: `http://127.0.0.1:${port}/github`,
        close: () =>
          new Promise((r) => {
            for (const socket of sockets) {
              socket.destroy();
            }
            server.close(() => r());
          }),
      });
    });
  });
}

// Plays back `answers`: an Error is thrown like a network failure, a number
// is answered as that HTTP status.
function playback(answers, body = {}) {
  const calls = [];
  const fetchImpl = async (url) => {
    calls.push(url);
    const answer = answers[calls.length - 1];
    if (answer instanceof Error) throw answer;
    return { ok: answer >= 200 && answer < 300, status: answer, json: async () => body };
  };
  return { calls, fetchImpl };
}

 * dirty index, and the bump must start from the newest state of the branch

function syncWithRemote(branch) {

/**
 * Push the release commit, retrying only genuine non-fast-forward races.
 * @param {string} branch
 * @param {number} maxAttempts
 */
function pushReleaseCommit(branch, maxAttempts = 3) {
  for (let attempt = 1; ; attempt++) {
    try {
      git(['push', 'origin', `HEAD:${branch}`]);
      return;
    } catch (error) {
      const output = gitErrorOutput(error);
      const kind = classifyPushFailure(output);
      if (kind === 'repository-rules') {
        throw new Error(
          `The push to '${branch}' was declined by a repository rule (GH006/GH013). ` +
            `Rebasing cannot change repository policy: allow the release bot to push to '${branch}' ` +
            `or release through a pull request.\n${output}`,
        );
      }
      if (kind === 'other' || attempt >= maxAttempts) {
        throw new Error(`git push failed after ${attempt} attempt(s):\n${output}`);
      }
      console.log(
        `Push rejected as non-fast-forward (attempt ${attempt}/${maxAttempts}); pulling with rebase and retrying...`,
      );
      try {
        git(['pull', '--rebase', 'origin', branch]);
      } catch (rebaseError) {
        try {
          git(['rebase', '--abort']);
        } catch {
          // Nothing to abort.
        }
        throw new Error(`git pull --rebase failed:\n${gitErrorOutput(rebaseError)}`);
      }
    }
  }

function main() {
  // Configure git
  git(['config', 'user.name', 'github-actions[bot]']);
  git(['config', 'user.email', 'github-actions[bot]@users.noreply.github.com']);

  // A detached checkout reports "HEAD"; fall back to the workflow's branch.
  let branch = '';
  try {
    branch = git(['rev-parse', '--abbrev-ref', 'HEAD']);
  } catch {
    // Unborn or detached HEAD: use the fallback below.
  }
  if (!branch || branch === 'HEAD') {
    branch = process.env.GITHUB_REF_NAME || 'main';
  }
  syncWithRemote(branch);

  let bumpType;
  let descriptions = [];

  if (mode === 'changeset') {
    // Changeset mode: get bump type from changesets
    const result = processChangesets();
    if (!result) {
      console.log('No changesets to process, exiting');
      setOutput('version_committed', 'false');
      setOutput('already_released', 'false');
      return;
    }
    bumpType = result.bumpType;
    descriptions = result.descriptions;
  } else if (mode === 'instant') {
    // Instant mode: use provided bump type
    if (!bumpTypeArg || !['major', 'minor', 'patch'].includes(bumpTypeArg)) {
      console.error(
        'Usage: node csharp/scripts/version-and-commit.mjs --mode instant --bump-type <major|minor|patch> [--description <desc>]',
      );
      process.exit(1);
    }
    bumpType = bumpTypeArg;
    if (description) {
      descriptions = [description];
    }
  } else {
    console.error('Invalid mode. Use --mode changeset or --mode instant');
    process.exit(1);
  }

  const current = getCurrentVersion();
  const newVersion = calculateNewVersion(current, bumpType);
  const tagName = `csharp-v${newVersion}`;

  // Check if this version was already released
  if (checkTagExists(newVersion)) {
    console.log(`Tag ${tagName} already exists`);
    setOutput('already_released', 'true');
    setOutput('new_version', newVersion);
    return;
  }

  // Update version in csproj
  updateCsproj(newVersion);

  // Update changelog if we have descriptions
  if (descriptions.length > 0) {
    updateChangelog(newVersion, descriptions);
  }

  // Remove changeset files (only in changeset mode)
  if (mode === 'changeset') {
    removeChangesetFiles();
  }

  // Stage all changed files; `-A` also records the removed changesets.
  const stagedPaths = [CSPROJ_PATH];
  if (existsSync(CHANGELOG_FILE)) {
    stagedPaths.push(CHANGELOG_FILE);
  }
  if (existsSync(LIBRARY_CSPROJ_PATH)) {
    stagedPaths.push(LIBRARY_CSPROJ_PATH);
  }
  git(['add', '--', ...stagedPaths]);
  git(['add', '-A', '--', CHANGESET_DIR]);

  // The csproj bump always changes a file, so an empty index means the
  // release write was lost: fail instead of publishing an uncommitted version.
  const staged = git(['diff', '--cached', '--name-only']);
  if (!staged) {
    throw new Error(
      `Nothing was staged for ${tagName}; refusing to publish a version that is not committed`,
    );
  }
  console.log(`Staged release files:\n${staged}`);

  // Commit changes
  const commitMsg = description
    ? `chore(csharp): release v${newVersion}\n\n${description}`
    : `chore(csharp): release v${newVersion}`;
  git(['commit', '-m', commitMsg]);
  console.log(`Committed version ${newVersion}`);

  pushReleaseCommit(branch);

  // Tag only after the commit reached the remote, so a rebase retry can never
  // leave the tag on an orphaned commit, and push exactly this tag.
  const tagMsg = description
    ? `C# Release v${newVersion}\n\n${description}`
    : `C# Release v${newVersion}`;
  git(['tag', '-a', tagName, '-m', tagMsg]);
  console.log(`Created tag ${tagName}`);
  git(['push', 'origin', tagName]);
  console.log(`Pushed release commit and tag ${tagName}`);

  setOutput('version_committed', 'true');
  setOutput('new_version', newVersion);
}

/**
 * Get current version from csproj
 * @returns {{major: number, minor: number, patch: number}}
 */
function getCurrentVersion() {
  const csproj = readFileSync(CSPROJ_PATH, 'utf-8');
  const match = csproj.match(/<Version>(\d+)\.(\d+)\.(\d+)<\/Version>/);

  if (!match) {
    console.error('Error: Could not parse version from csproj');
    process.exit(1);
  }

  return {
    major: parseInt(match[1], 10),
    minor: parseInt(match[2], 10),
    patch: parseInt(match[3], 10),
  };
}

/**
 * Calculate new version based on bump type
 * @param {{major: number, minor: number, patch: number}} current
 * @param {string} bumpType
 * @returns {string}
 */
function calculateNewVersion(current, bumpType) {
  const { major, minor, patch } = current;

  switch (bumpType) {
    case 'major':
      return `${major + 1}.0.0`;
    case 'minor':
      return `${major}.${minor + 1}.0`;
    case 'patch':
      return `${major}.${minor}.${patch + 1}`;
    default:
      throw new Error(`Invalid bump type: ${bumpType}`);
  }
}

/**
 * Update version in a csproj at the given path.
 * @param {string} path

function updateCsprojAt(path, newVersion) {

  csproj = csproj.replace(/<Version>[^<]+<\/Version>/, `<Version>${newVersion}</Version>`);
  writeFileSync(path, csproj, 'utf-8');
  console.log(`Updated ${path} to version ${newVersion}`);

/**
 * Parse a changeset file and extract its metadata
 * @param {string} filePath
 * @returns {{type: string, description: string} | null}
 */
function parseChangeset(filePath) {
  try {
    const content = readFileSync(filePath, 'utf-8');

    // Extract version type - support both quoted and unquoted package names
    const versionTypeRegex = new RegExp(
      `^['"]?${PACKAGE_NAME.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}['"]?:\\s+(major|minor|patch)`,
      'm',
    );
    const versionTypeMatch = content.match(versionTypeRegex);

    if (!versionTypeMatch) {
      console.warn(`Warning: Could not parse version type from ${filePath}`);
      return null;
    }

    // Extract description
    const parts = content.split('---');
    const desc = parts.length >= 3 ? parts.slice(2).join('---').trim() : '';

    return {
      type: versionTypeMatch[1],
      description: desc,
    };
  } catch (error) {
    console.warn(`Warning: Failed to parse ${filePath}: ${error.message}`);
    return null;
  }
}

/**
 * Get the highest priority bump type
 * @param {string[]} types
 * @returns {string}
 */
function getHighestBumpType(types) {
  let highest = 'patch';
  for (const type of types) {
    if (BUMP_PRIORITY[type] > BUMP_PRIORITY[highest]) {
      highest = type;
    }
  }
  return highest;
}

/**
 * Get changeset files from .changeset directory
 * @returns {string[]}
 */
function getChangesetFiles() {
  if (!existsSync(CHANGESET_DIR)) {
    return [];
  }
  return readdirSync(CHANGESET_DIR).filter(
    (file) => file.endsWith('.md') && file !== 'README.md' && file !== 'config.json',
  );
}

/**
 * Process changesets and return bump type and descriptions
 * @returns {{bumpType: string, descriptions: string[]} | null}
 */
function processChangesets() {
  const files = getChangesetFiles();

  if (files.length === 0) {
    console.log('No changeset files found');
    return null;
  }

  console.log(`Found ${files.length} changeset file(s)`);

  const parsedChangesets = [];
  const invalid = [];
  for (const file of files) {
    const filePath = join(CHANGESET_DIR, file);
    const parsed = parseChangeset(filePath);
    if (parsed) {
      parsedChangesets.push({
        file,
        filePath,
        ...parsed,
      });
    } else {
      invalid.push(file);
    }
  }

  // Every changeset is deleted after the release, so one that does not parse
  // would lose its bump and its description on a green run (issue #134).
  if (invalid.length > 0) {
    console.error(
      `::error title=Invalid changeset::Cannot parse ${invalid.join(', ')} in ${CHANGESET_DIR}. ` +
        `How to fix: start it with ---, '${PACKAGE_NAME}': major|minor|patch, --- and a description.`,
    );
    process.exit(1);
  }

  const bumpTypes = parsedChangesets.map((c) => c.type);
  const highestBumpType = getHighestBumpType(bumpTypes);
  const descriptions = parsedChangesets.filter((c) => c.description).map((c) => c.description);

  console.log(`Bump types found: ${[...new Set(bumpTypes)].join(', ')}`);
  console.log(`Using highest: ${highestBumpType}`);

  return {
    bumpType: highestBumpType,
    descriptions,
  };
}

/**
 * Insert a release entry above the newest one, separated by exactly one blank
 * line. Splicing an entry that started with "\n" after the existing blank
 * lines used to add one more empty line below the header on every release.
 * @param {string | null} changelog existing CHANGELOG.md content, or null
 * @param {string} entry the "## [version] - date" section
 * @returns {string}
 */
export function insertChangelogEntry(changelog, entry) {
  const section = entry.trim();
  const content = changelog ?? CHANGELOG_HEADER;
  const match = /^## \[/m.exec(content);
  if (!match) {
    return `${content.trimEnd()}\n\n${section}\n`;
  }
  const before = content.slice(0, match.index).trimEnd();
  const after = content.slice(match.index);
  return `${before}\n\n${section}\n\n${after}`;
}

/** The text of a `~1{hex}` literal. */
export function decodeReferenceLiteral(literal) {
  const match = /^~([0-9]+)\{([0-9a-fA-F]*)\}$/.exec(literal);
  if (!match || match[1] !== '1') {
    throw new MalformedMessageError(`invalid or unsupported reference literal: ${literal}`);
  }

  if (hex.length % 2 !== 0) {
    throw new MalformedMessageError('reference literal requires an even number of hex digits');

  } catch {

  }
}

/** Writes a reference so that it parses back as exactly `text`. */
export function formatReference(text) {
  if (typeof text !== 'string') throw new TypeError(`a reference is a string, got ${typeof text}`);
  if (text === '' || CONTROL.test(text)) return encodeReferenceLiteral(text);
  if (!LITERAL_PREFIX.test(text) && !text.startsWith('#') && !NEEDS_QUOTES.test(text)) return text;
  const first = text[0];
  let best = null;
  for (const quote of QUOTES) {
    if (quote === first) continue;
    let longest = 0;
    let current = 0;
    for (const c of text) {
      current = c === quote ? current + 1 : 0;
      if (current > longest) longest = current;
    }
    const count = (longest + 1) | 1;
    if (best === null || count < best.count) best = { quote, count };
  }
  const delimiter = best.quote.repeat(best.count);
  return `${delimiter}${text}${delimiter}`;
}

const isSpace = (c) => c === ' ' || c === '\t';
const isLineBreak = (c) => c === '\n' || c === '\r';
const endsBareWord = (c) =>
  c === undefined || isSpace(c) || isLineBreak(c) || c === '(' || c === ')' || c === ':';

function runLength(text, start, quote) {
  let end = start;
  while (text[end] === quote) end += 1;
  return end - start;
}

/** Whether an even-run body holds something visible and balanced parentheses. */
function isSubstantiveBody(content) {
  let depth = 0;
  let visible = false;
  for (const c of content) {
    if (c === '(') depth += 1;
    if (c === ')') {
      depth -= 1;
      if (depth < 0) return false;
    }
    if (!isSpace(c) && !isLineBreak(c)) visible = true;
  }
  return visible && depth === 0;
}

/**
 * Reads the delimited reference opened at `start`: a run of N quotes opens it,
 * a run of R closes it when `R / N` is odd (its last N close it), and each 2N
 * inside are N literal quotes. An even opening run with no substantive body is
 * the empty reference.
 */
function readQuoted(text, start) {
  const quote = text[start];
  const count = runLength(text, start, quote);
  let position = start + count;
  let end = -1;
  for (;;) {
    const found = text.indexOf(quote, position);
    if (found === -1) break;
    const length = runLength(text, found, quote);
    position = found + length;
    if (length >= count && Math.floor(length / count) % 2 === 1) {
      end = position;
      break;
    }
  }
  const empty = count % 2 === 0 ? { value: '', end: start + count } : null;
  if (end === -1) {
    if (empty) return empty;
    throw new MalformedMessageError(`unterminated quoted reference at offset ${start}`);
  }
  const body = text.slice(start + count, end);
  let value = '';
  let at = 0;
  for (;;) {
    const found = body.indexOf(quote, at);
    if (found === -1) break;
    const length = runLength(body, found, quote);
    const escaped = Math.floor(length / (2 * count)) * count;
    const closes = found + length === body.length ? count : 0;
    value += body.slice(at, found) + quote.repeat(length - escaped - closes);
    at = found + length;
  }
  value += body.slice(at);
  if (empty && !isSubstantiveBody(value)) return empty;
  return { value, end };
}

  constructor(text) {
    this.text = text;

  fail(expected) {

    if (LITERAL_PREFIX.test(text.slice(this.at, this.at + 24))) {

  }

    return this.text[this.at] === '(' ? this.parenthesized() : this.reference();
  }

  /** One top-level line: `link`, `a b c` or `id: a b`. */
  line() {
    const values = [];
    let id = null;
    for (;;) {
      this.skipSpace();
      const c = this.text[this.at];
      if (c === undefined || isLineBreak(c)) break;
      if (c === ')') this.fail('a link or a reference');

        this.at += 1;
        id = values.pop();
        continue;
      }
      values.push(this.value());
    }
    if (id === null && values.length === 1) return values[0];
    return { id, values };
  }
}

/** A transport answering from a script: `answer(text) => reply text`. */
function scripted(answer, capabilities = tcpServerCapabilities()) {
  const sent = [];
  return {
    sent,
    capabilities,
    closed: 0,
    async send(text, scope) {
      sent.push(text);
      if (scope?.signal?.aborted) throw new Error('scope ignored');
      return answer(text, sent.length);
    },
    async close() {
      this.closed += 1;
    },
  };
}

/** `[{ request, reply }]` of the golden conversation both servers replay. */
async function goldenConversation() {
  const lines = (await readFile(golden, 'utf8'))
    .split('\n')
    .filter((line) => line !== '' && !line.startsWith('#'));
  const exchanges = [];
  for (const line of lines) {

    else exchanges.at(-1).reply.push(line);
  }
  return exchanges.map(({ request, reply }) => ({ request, reply: reply.join('\n') }));

/**
 * `[{ connection, kind, text, reply }]` of a conversation over two connections:
 * `kind` is `'request'` (`A> text`, then its reply lines) or `'push'`
 * (`A< (notify: n)`, then its change lines, all in `reply`).
 */
async function twoConnectionConversation(file) {
  const lines = (await readFile(file, 'utf8'))
    .split('\n')
    .filter((line) => line !== '' && !line.startsWith('#'));
  const steps = [];
  for (const line of lines) {
    const step = /^([AB])([<>]) (.*)$/.exec(line);
    if (step) {
      const [, connection, direction, text] = step;
      steps.push(
        direction === '>'
          ? { connection, kind: 'request', text, reply: [] }
          : { connection, kind: 'push', text, reply: [text] },
      );
    } else {
      steps.at(-1).reply.push(line);
    }
  }
  return steps.map((step) => ({ ...step, reply: step.reply.join('\n') }));
}

/** A server process backend: a fresh database in a directory with spaces. */
function processBackend(name, path, capabilities) {
  const [command, args] = path.endsWith('.dll') ? ['dotnet', [path]] : [path, []];
  return {
    name,
    process: true,
    persistent: true,
    async store() {
      const dir = await mkdtemp(join(tmpdir(), `links client ${name} `));
      const db = join(dir, 'conformance db.links');
      const options = { command, args, db, capabilities, timeout: 60_000 };
      return {
        db,
        options,
        /** A client on a server process of its own, stopped by `close()`. */
        open: (extra = {}) => connectProcess({ ...options, ...extra }),
        start: (extra = {}) => startServer({ ...options, ...extra }),
        dispose: () => rm(dir, { recursive: true, force: true }),
      };
    },
  };
}

/** The in-process WebAssembly store; it lives as long as the `Clink`. */
async function wasmBackend(dir) {
  const module = await import(pathToFileURL(join(dir, 'clink_wasm.js')).href);
  module.initSync({ module: readFileSync(join(dir, 'clink_wasm_bg.wasm')) });
  return {
    name: 'wasm',
    process: false,
    persistent: false,
    async store() {
      const clink = new module.Clink();
      return {
        clink,
        /** A client borrowing the store: closing it leaves the store alive. */
        open: async (extra = {}) => createWasmClient(clink, extra),
        dispose: async () => clink.free(),

    ) {
      throw new Error(
        `${backend.name}: no executable at ${process.env[`CLINK_${backend.name.toUpperCase()}`]}`,
      );
    }
  }
  return backends;
}

// ---------------------------------------------------------------- helpers

const indexes = (links) => links.map((link) => link.index);

/** `{ index: { source, target } }` of every link, for readback comparisons. */
async function snapshot(client) {
  return Object.fromEntries(
    (await client.each()).map(({ index, source, target }) => [index, `${source} ${target}`]),
  );
}

/** Reopens the store with a client that shares nothing with the previous one. */
async function fresh(store, previous) {
  await previous?.close();
  return store.open();
}

async function using(store, body) {
  let client = await store.open();
  try {
    return await body(client, async () => {
      client = await fresh(store, client);
      return client;
    });
  } finally {
    await client.close();
  }
}

// ---------------------------------------------------------------- harness

function withTimeout(promise, ms, label) {
  let timer;
  const timeout = new Promise((_, reject) => {
    timer = setTimeout(() => reject(new Error(`${label} took longer than ${ms} ms`)), ms);
  });
  return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
}

async function main() {
  const backends = await backendsFromEnvironment();
  if (backends.length === 0) {
    console.error('set CLINK_RUST, CLINK_CSHARP and/or CLINK_WASM; see the top of this file');
    process.exit(2);
  }
  console.log(
    `# link-cli-client conformance on ${runtime}: ${backends.map((backend) => backend.name).join(', ')}`,
  );
  let failed = 0;
  let passed = 0;
  let skipped = 0;
  for (const backend of backends) {
    for (const scenario of scenarios) {
      const label = `${backend.name}: ${scenario.name}`;
      if (
        (scenario.persistentOnly && !backend.persistent) ||
        (scenario.processOnly && !backend.process)
      ) {
        skipped += 1;
        console.log(
          `skip - ${label} (${scenario.persistentOnly ? 'not persistent' : 'no server process'})`,
        );
        continue;
      }
      const started = performance.now();
      const store = await backend.store();
      try {
        await withTimeout(scenario.run(store, backend), SCENARIO_TIMEOUT_MS, label);
        passed += 1;
        console.log(`ok - ${label} (${Math.round(performance.now() - started)} ms)`);
      } catch (error) {
        failed += 1;
        console.log(`not ok - ${label}`);
        console.log(String(error?.stack ?? error).replace(/^/gm, '  # '));
      } finally {
        await store.dispose();
      }
    }
  }
  console.log(`# ${passed} passed, ${failed} failed, ${skipped} skipped`);
  process.exit(failed === 0 ? 0 : 1);
}

const golden = new URL('../../../../docs/protocol/links-operations.txt', import.meta.url);

/** `[{ request, reply }]` of the golden conversation both servers replay. */
async function goldenConversation() {
  const lines = (await readFile(golden, 'utf8'))
    .split('\n')
    .filter((line) => line !== '' && !line.startsWith('#'));
  const exchanges = [];
  for (const line of lines) {
    if (line.startsWith('> ')) exchanges.push({ request: line.slice(2), reply: [] });
    else exchanges.at(-1).reply.push(line);
  }
  return exchanges.map(({ request, reply }) => ({ request, reply: reply.join('\n') }));
}

const subscriptionsGolden = new URL('../../../../docs/protocol/subscriptions.txt', import.meta.url);
const tripletsGolden = new URL('../../../../docs/protocol/triplets.txt', import.meta.url);

/**
 * `[{ connection, kind, text, reply }]` of a conversation over two connections:
 * `kind` is `'request'` (`A> text`, then its reply lines) or `'push'`
 * (`A< (notify: n)`, then its change lines, all in `reply`).
 */
async function twoConnectionConversation(file) {
  const lines = (await readFile(file, 'utf8'))
    .split('\n')
    .filter((line) => line !== '' && !line.startsWith('#'));
  const steps = [];
  for (const line of lines) {
    const step = /^([AB])([<>]) (.*)$/.exec(line);
    if (step) {
      const [, connection, direction, text] = step;
      steps.push(
        direction === '>'
          ? { connection, kind: 'request', text, reply: [] }
          : { connection, kind: 'push', text, reply: [text] },
      );
    } else {
      steps.at(-1).reply.push(line);
    }
  }
  return steps.map((step) => ({ ...step, reply: step.reply.join('\n') }));
}

// ---------------------------------------------------------------- backends

/** A server process backend: a fresh database in a directory with spaces. */
function processBackend(name, path, capabilities) {
  const [command, args] = path.endsWith('.dll') ? ['dotnet', [path]] : [path, []];
  return {
    name,
    process: true,
    persistent: true,
    async store() {
      const dir = await mkdtemp(join(tmpdir(), `links client ${name} `));
      const db = join(dir, 'conformance db.links');
      const options = { command, args, db, capabilities, timeout: 60_000 };
      return {
        db,
        options,
        /** A client on a server process of its own, stopped by `close()`. */
        open: (extra = {}) => connectProcess({ ...options, ...extra }),
        start: (extra = {}) => startServer({ ...options, ...extra }),
        dispose: () => rm(dir, { recursive: true, force: true }),
      };
    },
  };
}

/** The in-process WebAssembly store; it lives as long as the `Clink`. */
async function wasmBackend(dir) {
  const module = await import(pathToFileURL(join(dir, 'clink_wasm.js')).href);
  module.initSync({ module: readFileSync(join(dir, 'clink_wasm_bg.wasm')) });
  return {
    name: 'wasm',
    process: false,
    persistent: false,
    async store() {
      const clink = new module.Clink();
      return {
        clink,
        /** A client borrowing the store: closing it leaves the store alive. */
        open: async (extra = {}) => createWasmClient(clink, extra),
        dispose: async () => clink.free(),
      };
    },
  };
}

async function backendsFromEnvironment() {
  const backends = [];
  const { CLINK_RUST, CLINK_CSHARP, CLINK_WASM } = process.env;
  // Stated, not discovered: the protocol does not say which server answers.
  if (CLINK_RUST) backends.push(processBackend('rust', CLINK_RUST, { duplicateLinks: true }));
  if (CLINK_CSHARP)
    backends.push(processBackend('csharp', CLINK_CSHARP, { duplicateLinks: false }));
  if (CLINK_WASM) backends.push(await wasmBackend(CLINK_WASM));
  for (const backend of backends) {
    if (
      backend.name !== 'wasm' &&
      !existsSync(process.env[`CLINK_${backend.name.toUpperCase()}`])
    ) {
      throw new Error(
        `${backend.name}: no executable at ${process.env[`CLINK_${backend.name.toUpperCase()}`]}`,
      );
    }
  }
  return backends;
}

// ---------------------------------------------------------------- helpers

const indexes = (links) => links.map((link) => link.index);

/** `{ index: { source, target } }` of every link, for readback comparisons. */
async function snapshot(client) {
  return Object.fromEntries(
    (await client.each()).map(({ index, source, target }) => [index, `${source} ${target}`]),
  );
}

/** Reopens the store with a client that shares nothing with the previous one. */
async function fresh(store, previous) {
  await previous?.close();
  return store.open();
}

async function withClient(store, body) {
  let client = await store.open();
  try {
    return await body(client, async () => {
      client = await fresh(store, client);
      return client;
    });
  } finally {
    await client.close();
  }
}

        `it's "quoted" \`thrice\``,
        "''",
        '(parens): and colon',
        'line\nbreak\r\nand\ttab',
        '~1{41}',
        '#hash',
        '  spaces around  ',
        'x'.repeat(2000),
      ];
      await withClient(store, async (client, reopen) => {
        const links = [];
        for (const [i, name] of names.entries()) {
          const link = await client.create(BigInt(1000 + i), BigInt(1000 + i));
          links.push(link);
          await client.setName(link.index, name);
        }
        const check = async (reader) => {
          for (const [i, name] of names.entries()) {
            assert.equal(await reader.getName(links[i].index), name, JSON.stringify(name));
            assert.equal(await reader.getByName(name), links[i].index, JSON.stringify(name));
          }
          const exported = await reader.exportLinks();
          assert.deepEqual(new Set(exported.map((link) => link.index)), new Set(names));
        };
        await check(client);
        await check(await reopen());
      });
    },
  },
  {
    name: 'empty names, renaming and moving names',
    async run(store) {
      await withClient(store, async (client, reopen) => {
        const a = await client.create(5n, 5n);
        const b = await client.create(6n, 6n);
        await client.setName(a.index, '');
        assert.equal(await client.getName(a.index), '');
        assert.equal(await client.getByName(''), a.index);
        await client.setName(a.index, 'moving');
        await client.setName(b.index, 'moving');
        assert.equal(await client.getByName('moving'), b.index);
        assert.equal(await client.getName(a.index), null);
        await client.removeName(b.index);
        const reader = await reopen();
        assert.equal(await reader.getByName('moving'), null);
        assert.equal(await reader.getName(b.index), null);
      });
    },
  },
  {
    name: 'limits: oversized requests are refused before sending',
    async run(store) {
      const client = await store.open({ maxRequestBytes: 64 });

        await assert.rejects(client.getByName('n'.repeat(100)), {
          name: 'LimitExceededError',
          outcome: 'not-sent',
        });
        assert.equal(await client.count(), 0n);
      } finally {
        await client.close();
      }
    },
  },
  {
    name: 'incompatible protocol: a binary-only server is refused up front',
    processOnly: true,
    async run(store) {
      const server = await store.start({ args: [...store.options.args, '--protocol', 'binary'] });
      try {
        await assert.rejects(connectTcp(server.address, { timeout: 10_000 }), {
          name: 'ProtocolMismatchError',
          message: /does not accept this protocol/,
        });
      } finally {
        await server.close();
      }
    },
  },
  {
    name: 'ownership: a client stops only the server process it started',
    processOnly: true,
    async run(store) {
      const owning = await store.open();
      const owned = owning.transport.server;
      await owning.close();
      const exit = await owned.exited;
      assert.ok(exit.code === 0 || exit.signal === 'SIGINT', JSON.stringify(exit));

      const server = await store.start();
      try {
        const borrowing = await connectTcp(server.address);
        await borrowing.create(1n, 1n);
        await borrowing.close();
        // The server it connected to outlives it.
        const other = await connectTcp(server.address);
        assert.equal(await other.count(), 1n);
        await other.close();
      } finally {
        await server.close();
      }
    },
  },

function withTimeout(promise, ms, label) {
  let timer;
  const timeout = new Promise((_, reject) => {
    timer = setTimeout(() => reject(new Error(`${label} took longer than ${ms} ms`)), ms);
  });
  return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
}

async function main() {
  const backends = await backendsFromEnvironment();
  if (backends.length === 0) {
    console.error('set CLINK_RUST, CLINK_CSHARP and/or CLINK_WASM; see the top of this file');
    process.exit(2);
  }
  console.log(
    `# link-cli-client conformance on ${runtime}: ${backends.map((backend) => backend.name).join(', ')}`,
  );
  let failed = 0;
  let passed = 0;
  let skipped = 0;
  for (const backend of backends) {
    for (const scenario of scenarios) {
      const label = `${backend.name}: ${scenario.name}`;
      if (
        (scenario.persistentOnly && !backend.persistent) ||
        (scenario.processOnly && !backend.process)
      ) {
        skipped += 1;
        console.log(
          `skip - ${label} (${scenario.persistentOnly ? 'not persistent' : 'no server process'})`,
        );
        continue;
      }
      const started = performance.now();
      const store = await backend.store();
      try {
        await withTimeout(scenario.run(store, backend), SCENARIO_TIMEOUT_MS, label);
        passed += 1;
        console.log(`ok - ${label} (${Math.round(performance.now() - started)} ms)`);
      } catch (error) {
        failed += 1;
        console.log(`not ok - ${label}`);
        console.log(String(error?.stack ?? error).replace(/^/gm, '  # '));
      } finally {
        await store.dispose();
      }
    }
  }
  console.log(`# ${passed} passed, ${failed} failed, ${skipped} skipped`);
  process.exit(failed === 0 ? 0 : 1);
}

await main();
